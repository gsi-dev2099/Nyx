PL/pgSQL: convert_presale_to_order PostgreSQL 
CREATE OR REPLACE FUNCTION lead_service.convert_presale_to_order(
    p_id_presale bigint, 
    p_id_cmpg bigint, 
    p_id_user bigint, 
    p_id_src bigint DEFAULT 1
) RETURNS bigint AS $$
DECLARE
    v_presale   lead_service.lead_pre_sale%ROWTYPE;
    v_lead_id   BIGINT;
    v_status_id BIGINT;
BEGIN
    -- 1. Obtener status inicial OPEN
    SELECT id_sts INTO v_status_id 
    FROM lead_service.lead_status
    WHERE category = 'OPEN' 
    ORDER BY order_index LIMIT 1;

    -- 2. Insertar Lead formal
    INSERT INTO lead_service.lead
        (id_cmpg, id_src, first_name, last_name, phone, current_status_id, assigned_user_id)
    VALUES
        (p_id_cmpg, p_id_src, v_presale.first_name, v_presale.last_name, v_presale.phone, v_status_id, p_id_user)
    RETURNING id_lead INTO v_lead_id;

    RETURN v_lead_id;
END;
$$ LANGUAGE plpgsql;
C#: ValidationHelper.cs Blazor Client 
public static class ValidationHelper
{
    public static bool IsValidDni(string dni)
    {
        if (string.IsNullOrWhiteSpace(dni) || dni.Length != 9) return false;
        var numberPart = dni.Substring(0, 8);
        var letter = char.ToUpper(dni[8]);
        
        if (!long.TryParse(numberPart, out long number)) return false;
        
        const string letters = "TRWAGMYFPDXBNJZSQVHLCKE";
        return letter == letters[(int)(number % 23)];
    }
}
